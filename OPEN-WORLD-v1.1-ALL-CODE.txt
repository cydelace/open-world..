const path = require('path');
const http = require('http');
const express = require('express');
const { Server } = require('socket.io');

const app = express();
const server = http.createServer(app);
const io = new Server(server, { cors: { origin: '*' } });
const PORT = process.env.PORT || 3000;

app.use(express.static(path.join(__dirname, 'public')));
app.get('/health', (_req, res) => res.json({ ok: true, players: io.engine.clientsCount }));

const players = new Map();

io.on('connection', (socket) => {
  socket.on('join-world', (payload = {}) => {
    const name = String(payload.name || 'Driver').trim().slice(0, 16) || 'Driver';
    const player = {
      id: socket.id,
      name,
      x: Number(payload.x) || 2.8,
      z: Number(payload.z) || 10,
      rotation: 0,
      inCar: false,
      carX: 0,
      carZ: 5,
      carRotation: 0
    };
    players.set(socket.id, player);
    socket.emit('world-state', [...players.values()].filter((p) => p.id !== socket.id));
    socket.broadcast.emit('player-joined', player);
  });

  socket.on('player-move', (state = {}) => {
    const current = players.get(socket.id);
    if (!current) return;
    // Accept only the small set of numeric fields used by the game.
    for (const key of ['x', 'z', 'rotation', 'carX', 'carZ', 'carRotation']) {
      if (Number.isFinite(state[key])) current[key] = state[key];
    }
    current.inCar = Boolean(state.inCar);
    io.emit('player-moved', current);
  });

  socket.on('disconnect', () => {
    players.delete(socket.id);
    socket.broadcast.emit('player-left', socket.id);
  });
});

server.listen(PORT, '0.0.0.0', () => {
  console.log(`Open World is running at http://localhost:${PORT}`);
});

function collidesWithBuilding(x, z, radius = 0.8) {
  return worldColliders.some((b) => Math.abs(x - b.x) < b.halfW + radius && Math.abs(z - b.z) < b.halfD + radius);

function enterOrExitCar() {
  if (!gameStarted) return;
  if (player.inCar) {
    const sideX = Math.cos(player.car.rotation.y) * 2.2;
    const sideZ = Math.sin(player.car.rotation.y) * 2.2;
    let exitX = player.car.position.x + sideX;
    let exitZ = player.car.position.z + sideZ;
    if (collidesWithBuilding(exitX, exitZ, 0.7)) { exitX += sideX * 1.5; exitZ += sideZ * 1.5; }
    player.x = exitX;
    player.z = exitZ;
    player.rotation = player.car.rotation.y;
    player.inCar = false;
    player.speed = 0;
    player.mesh.visible = true;
    showToast('You are on foot. Find another ride!');
  } else {
    let nearest = null;
    let nearestDistance = Infinity;
    for (const car of vehicleFleet) {
      const distance = Math.hypot(player.x - car.position.x, player.z - car.position.z);
      if (distance < nearestDistance) { nearest = car; nearestDistance = distance; }
    }
    if (nearest && nearestDistance < 4.8) {
      player.car = nearest;
      player.inCar = true;
      player.mesh.visible = false;
      player.car.position.y = 0;
      player.speed = 0;
      showToast(`Driving: ${player.car.userData.vehicleName || 'CITY CAR'}`);
    }
  }
}

const clock = new THREE.Clock();
function update(dt) {
  const forwardInput = (keyDown('KeyW') || keyDown('ArrowUp') ? 1 : 0) - (keyDown('KeyS') || keyDown('ArrowDown') ? 1 : 0);
  const sideInput = (keyDown('KeyD') || keyDown('ArrowRight') ? 1 : 0) - (keyDown('KeyA') || keyDown('ArrowLeft') ? 1 : 0);

  if (!player.inCar) {
    const cameraForwardX = Math.sin(cameraYaw);
    const cameraForwardZ = -Math.cos(cameraYaw);
    const rightX = Math.cos(cameraYaw);
    const rightZ = Math.sin(cameraYaw);
    let moveX = cameraForwardX * forwardInput + rightX * sideInput;
    let moveZ = cameraForwardZ * forwardInput + rightZ * sideInput;
    const length = Math.hypot(moveX, moveZ);
    if (length > 0.001) {
      moveX /= length; moveZ /= length;
      const run = keyDown('ShiftLeft') || keyDown('ShiftRight');
      const pace = run ? 7.8 : 4.6;
      const nextX = player.x + moveX * pace * dt;
      const nextZ = player.z + moveZ * pace * dt;
      if (!collidesWithBuilding(nextX, player.z, 0.55)) player.x = nextX;
      if (!collidesWithBuilding(player.x, nextZ, 0.55)) player.z = nextZ;
      player.rotation = Math.atan2(moveX, -moveZ);
      player.walkPhase += dt * (run ? 12 : 8);
      player.mesh.position.y = Math.abs(Math.sin(player.walkPhase)) * 0.055;
    } else {
      player.mesh.position.y = THREE.MathUtils.damp(player.mesh.position.y, 0, 8, dt);
    }
    player.mesh.position.x = player.x;
    player.mesh.position.z = player.z;
    player.mesh.rotation.y = player.rotation;
    let closestCar = null;
    let closestCarDistance = Infinity;
    for (const car of vehicleFleet) {
      const distance = Math.hypot(player.x - car.position.x, player.z - car.position.z);
      if (distance < closestCarDistance) { closestCar = car; closestCarDistance = distance; }
    }
    const nearCar = closestCarDistance < 4.8;
    promptEl.classList.toggle('hidden', !nearCar);
    if (nearCar && closestCar) promptEl.innerHTML = `Press <kbd>E</kbd> to enter ${closestCar.userData.vehicleName || 'vehicle'}`;
  } else {
    promptEl.classList.remove('hidden');
    promptEl.innerHTML = 'Press <kbd>E</kbd> to exit vehicle';
    const acceleration = 17;
    const maxSpeed = 28;
    if (forwardInput > 0) player.speed = Math.min(maxSpeed, player.speed + acceleration * dt);
    else if (forwardInput < 0) player.speed = Math.max(-10, player.speed - acceleration * 0.72 * dt);
    else player.speed *= Math.pow(0.28, dt);
    if (Math.abs(player.speed) < 0.12) player.speed = 0;
    const steeringStrength = THREE.MathUtils.clamp(Math.abs(player.speed) / 9, 0, 1);
    player.car.rotation.y += sideInput * steeringStrength * Math.sign(player.speed || 1) * 1.55 * dt;
    const dx = Math.sin(player.car.rotation.y) * player.speed * dt;
    const dz = -Math.cos(player.car.rotation.y) * player.speed * dt;
    const nextX = THREE.MathUtils.clamp(player.car.position.x + dx, -116, 116);
    const nextZ = THREE.MathUtils.clamp(player.car.position.z + dz, -116, 116);
    if (!collidesWithBuilding(nextX, nextZ, 1.35)) {
      player.car.position.x = nextX;
      player.car.position.z = nextZ;
    } else {
      player.speed *= -0.18;
    }
    for (const wheel of player.car.userData.wheels) wheel.rotation.x += player.speed * dt * 1.7;
    player.x = player.car.position.x;
    player.z = player.car.position.z;
    player.rotation = player.car.rotation.y;
  }

  // Ambient vehicles cruise along the marked city lanes.
  for (const traffic of trafficCars) {
    const posKey = traffic.axis === 'x' ? 'x' : 'z';
    traffic.mesh.position[posKey] += traffic.direction * traffic.speed * dt;
    if (traffic.mesh.position[posKey] > 118) traffic.mesh.position[posKey] = -118;
    if (traffic.mesh.position[posKey] < -118) traffic.mesh.position[posKey] = 118;
    for (const wheel of traffic.mesh.userData.wheels) wheel.rotation.x += traffic.speed * dt * 1.4;
  }
  // Pedestrians follow simple sidewalk routes and turn around at district edges.
  for (const ped of pedestrians) {
    const posKey = ped.axis === 'x' ? 'x' : 'z';
    ped.mesh.position[posKey] += ped.direction * ped.speed * dt;
    if (ped.mesh.position[posKey] > 108) ped.mesh.position[posKey] = -108;
    if (ped.mesh.position[posKey] < -108) ped.mesh.position[posKey] = 108;
    ped.mesh.position.y = Math.abs(Math.sin(performance.now() * 0.004 + ped.phase)) * 0.035;
  }
  // Pulse the active mission ring so it is easy to spot in the city.
  checkpointMeshes.forEach((marker, index) => {
    marker.group.visible = index === missionIndex;
    if (index === missionIndex) {
      marker.ring.rotation.z += dt * 0.55;
      marker.group.position.y = 0.08 + Math.sin(performance.now() * 0.0025) * 0.12;
    }
  });
  const missionTarget = missionStops[missionIndex];
  if (missionTarget) {
    const targetPosition = player.inCar ? player.car.position : player.mesh.position;
    if (Math.hypot(targetPosition.x - missionTarget.x, targetPosition.z - missionTarget.z) < 6.5) {
      credits += missionTarget.reward;
      missionIndex += 1;
      creditsEl.textContent = `$${credits}`;
      refreshMissionHud();
      showToast(`Checkpoint reached! +$${missionTarget.reward}`);
    }
  }
  // Driving into small street props knocks them aside for a small arcade score bonus.
  if (player.inCar) {
    for (const prop of breakableProps) {
      if (!prop.active) continue;
      if (Math.hypot(player.car.position.x - prop.x, player.car.position.z - prop.z) < 1.9) {
        prop.active = false;
        scene.remove(prop.mesh);
        breakableScore += 25;
        scoreEl.textContent = String(breakableScore);
        showToast('Street prop cleared · +25 CITY SCORE');
      }
    }
  }

  speedEl.textContent = String(Math.round(Math.abs(player.speed) * 3.6));
  updateCamera(dt);
  if (socket?.connected && performance.now() - lastNetworkSend > 100) {
    lastNetworkSend = performance.now();
    socket.emit('player-move', {
      x: player.x, z: player.z, rotation: player.rotation, inCar: player.inCar,
      carX: player.car.position.x, carZ: player.car.position.z, carRotation: player.car.rotation.y
    });
  }
}

function updateCamera(dt) {
  const target = player.inCar ? player.car.position : player.mesh.position;
  const height = player.inCar ? 1.35 : 1.55;
  const horizontal = cameraDistance * Math.cos(cameraPitch);
  const vertical = cameraDistance * Math.sin(cameraPitch) + height;
  const desired = new THREE.Vector3(
    target.x + Math.sin(cameraYaw) * horizontal,
    target.y + vertical,
    target.z + Math.cos(cameraYaw) * horizontal
  );
  camera.position.lerp(desired, 1 - Math.exp(-5 * dt));
  camera.lookAt(target.x, target.y + height, target.z);
}

function animate() {
  requestAnimationFrame(animate);
  const dt = Math.min(clock.getDelta(), 0.04);
  if (gameStarted) update(dt);
  else {
    camera.position.set(29, 25, 34);
    camera.lookAt(0, 0, 0);
    camera.rotation.z = Math.sin(performance.now() * 0.00015) * 0.003;
  }
  renderer.render(scene, camera);
}

# OPEN WORLD — Living City Browser Prototype

A browser-based 3D open-world driving sandbox built with Three.js and Node.js. This upgraded starter adds a small living city while keeping the original solo and Socket.IO multiplayer entry points.

## Features

- Procedurally arranged 3D city blocks, roads, sidewalks, trees, lamps and a central plaza
- Third-person character movement, running, camera orbit and zoom
- Multiple enterable vehicles: city sedan, sports coupe, taxi and SUV
- Simple ambient traffic that cruises around the district
- Pedestrians walking along sidewalk routes
- Four city-tour checkpoints with in-game credit rewards
- Harmless arcade-style breakable street props that award city score
- Solo mode and a starter Socket.IO room for seeing other connected players
- Basic touch controls on smaller screens

## Requirements

- Node.js 18 or newer
- A modern desktop or mobile browser with WebGL support
- Internet access for the Three.js module and Google Fonts CDN

## Run the game

1. Install Node.js from https://nodejs.org/ if it is not already installed.
2. Extract this ZIP folder.
3. Open a terminal in the `open-world-starter` folder.
4. Run:

   ```bash
   npm install
   npm start
   ```

5. Open http://localhost:3000 in your browser.
6. Choose **Play Solo** or **Join Multiplayer**.

On Windows, you can also double-click `start.bat`; it installs the packages if needed and starts the server.

## Controls

- **W / A / S / D** or **arrow keys** — walk or drive
- **E** — enter the nearest parked vehicle or exit the current vehicle
- **Shift** — run
- **Mouse drag** — rotate the camera
- **Mouse wheel** — zoom the camera
- **Touch buttons** — move and enter/exit on supported mobile screens

## Multiplayer testing

For a local test, open the same address in two browser tabs and choose **Join Multiplayer** in both. For another device on the same Wi-Fi network, open the host computer's local network IP address with port `3000` (for example `http://192.168.1.10:3000`). The Node.js server must stay running.

Multiplayer currently relays player position and vehicle state. City traffic and pedestrians are simulated locally on each client. This is a prototype, not a public production service. Before a public release, add accounts, server-authoritative movement, input validation, rate limiting, moderation, persistence and deployment security.

## Current scope and next milestones

This is a playable foundation, not a finished commercial game. The vehicles and people are built from simple 3D shapes. The next upgrades could include detailed original models, better vehicle physics, a larger map, shops/interiors, mission variety, audio, save data, a minimap, and robust hosted multiplayer.
